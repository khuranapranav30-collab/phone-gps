function startGPS() { if (watchId !== undefined || client) { return; } if (!navigator.geolocation) { updateStatus("Location not supported", "#ef4444"); return; } updateStatus("Requesting phone location...", "#eab308"); // Request location first, before MQTT connection watchId = navigator.geolocation.watchPosition( (position) => { const data = { latitude: position.coords.latitude, longitude: position.coords.longitude, accuracy: position.coords.accuracy, timestamp: Date.now() }; document.getElementById("lat").innerText = data.latitude.toFixed(6); document.getElementById("lon").innerText = data.longitude.toFixed(6); document.getElementById("accuracy").innerText = data.accuracy.toFixed(1); updateStatus("GPS working. Connecting to MQTT...", "#eab308"); // Connect to MQTT only after getting location if (!client) { client = mqtt.connect(broker, { clientId: "phone_gps_" + Math.random().toString(16).slice(2) }); client.on("connect", () => { updateStatus("GPS and MQTT connected", "#22c55e"); }); client.on("error", (error) => { updateStatus("MQTT error: " + error.message, "#ef4444"); }); } if (client && client.connected) { client.publish( "motorcycle/gps", JSON.stringify(data), (error) => { if (error) { updateStatus("GPS working, MQTT publish failed", "#ef4444"); } else { updateStatus("Location published successfully", "#22c55e"); } } ); } }, (error) => { updateStatus("GPS error: " + error.message, "#ef4444"); watchId = undefined; }, { enableHighAccuracy: true, maximumAge: 1000, timeout: 15000 } ); }
